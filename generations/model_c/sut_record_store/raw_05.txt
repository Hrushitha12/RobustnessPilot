package pilot.records;

import org.junit.jupiter.api.Test;
import java.util.Map;
import java.util.LinkedHashMap;

public class RecordStore_ModelC_CoT_RobustnessTest {

    @Test
    public void constructorInvalidCapacityZero() throws Exception {
        new RecordStore(0, RecordType.STRING_ONLY);
    }

    @Test
    public void constructorInvalidCapacityNegative() throws Exception {
        new RecordStore(-5, RecordType.STRING_ONLY);
    }

    @Test
    public void constructorInvalidCapacityExceedsMax() throws Exception {
        new RecordStore(101, RecordType.STRING_ONLY);
    }

    @Test
    public void constructorValidCapacityMin() throws Exception {
        new RecordStore(1, RecordType.STRING_ONLY);
    }

    @Test
    public void constructorValidCapacityMax() throws Exception {
        new RecordStore(100, RecordType.STRING_ONLY);
    }

    @Test
    public void constructorNullType() throws Exception {
        new RecordStore(50, null);
    }

    @Test
    public void putNullKey() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.STRING_ONLY);
        store.put(null, "value");
    }

    @Test
    public void putBlankKey() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.STRING_ONLY);
        store.put("  ", "value");
    }

    @Test
    public void putKeyExceedsMaxLength() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.STRING_ONLY);
        store.put("a".repeat(33), "value");
    }

    @Test
    public void putNumericInvalidValue() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.NUMERIC_ONLY);
        store.putNumeric("key", null);
    }

    @Test
    public void putNumericInvalidFormat() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.NUMERIC_ONLY);
        store.putNumeric("key", "3.14");
    }

    @Test
    public void putNumericOutOfRange() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.NUMERIC_ONLY);
        store.putNumeric("key", String.valueOf(MAX_NUMERIC + 1));
    }

    @Test
    public void putOnNumericOnlyStore() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.NUMERIC_ONLY);
        store.put("key", "value");
    }

    @Test
    public void putNumericOnStringOnlyStore() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.STRING_ONLY);
        store.putNumeric("key", "123");
    }

    @Test
    public void putAfterClose() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.STRING_ONLY);
        store.close();
        store.put("key", "value");
    }

    @Test
    public void putAllWithInvalidEntry() throws Exception {
        RecordStore store = new RecordStore(1, RecordType.STRING_ONLY);
        Map<String, String> entries = new LinkedHashMap<>();
        entries.put(null, "value");
        store.putAll(entries);
    }

    @Test
    public void putAllExceedsCapacity() throws Exception {
        RecordStore store = new RecordStore(2, RecordType.STRING_ONLY);
        Map<String, String> entries = new LinkedHashMap<>();
        entries.put("a", "1");
        entries.put("b", "2");
        entries.put("c", "3");
        store.putAll(entries);
    }
}
